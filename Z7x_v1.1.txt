local fovCircle
pcall(function()
	fovCircle=Drawing.new("Circle")
	fovCircle.Thickness=1.5
	fovCircle.NumSides=60
	fovCircle.Filled=false
	fovCircle.Radius=150

local function clean(p)
	local d=data[p]
	if not d then return end
	if d.highlight then pcall(function() d.highlight:Destroy() end) end
	if d.line then pcall(function() d.line:Remove() end) end
	data[p]=nil
end

local function makeESP(p)
	if p==LP or not p.Character then return end

	clean(p)

	local h=Instance.new("Highlight")
	h.Parent=p.Character
	h.FillTransparency=.45
	h.OutlineTransparency=0

	local l
	pcall(function()
		l=Drawing.new("Line")
		l.Thickness=1.5
		l.Visible=false
	end)

	data[p]={highlight=h,line=l}
end

local function setup(p)
	if p==LP then return end

	p.CharacterAdded:Connect(function()
		clean(p)
		if espOn then
			task.wait(.1)
			makeESP(p)
		end
	end)

	p.CharacterRemoving:Connect(function()
		clean(p)
	end)

	if espOn then
		makeESP(p)
	end
end

for _,p in ipairs(P:GetPlayers()) do
	setup(p)
end

P.PlayerAdded:Connect(setup)
P.PlayerRemoving:Connect(clean)

local function visible(part)
	if not LP.Character then return false end

	local params=RaycastParams.new()
	params.FilterType=Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances={LP.Character,part.Parent}

	return workspace:Raycast(
		CG.CFrame.Position,
		part.Position-CG.CFrame.Position,
		params
	)==nil
end

local function closest()
	local target
	local best=fovSize

	for _,p in ipairs(P:GetPlayers()) do
		if p~=LP and p.Character and LP.Character then
			local part=p.Character:FindFirstChild(aimPart)
			local root=LP.Character:FindFirstChild("HumanoidRootPart")

			if part and root then
				local dist=(root.Position-part.Position).Magnitude

				if dist<=lockDistance then
					local pos,on=CG:WorldToViewportPoint(part.Position)

					if on then
						local center=Vector2.new(
							CG.ViewportSize.X/2,
							CG.ViewportSize.Y/2
						)

						local d=(Vector2.new(pos.X,pos.Y)-center).Magnitude

						if d<best and visible(part) then
							best=d
							target=part
						end
					end
				end
			end
		end
	end

	return target
end

local fpsOn=false

fps.Activated:Connect(function()
	fpsOn=not fpsOn

	fps.Text=fpsOn and "SHOW FPS & PING: ON" or "SHOW FPS & PING: OFF"
	fps.TextColor3=fpsOn
		and Color3.fromRGB(100,255,150)
		or Color3.fromRGB(255,105,120)

	if fpsOn then
		if not fpsLabel then
			fpsLabel=Instance.new("TextLabel",sg)
			fpsLabel.Size=UDim2.fromOffset(125,40)
			fpsLabel.Position=UDim2.fromOffset(15,15)
			fpsLabel.BackgroundColor3=Color3.fromRGB(15,15,20)
			fpsLabel.TextColor3=Color3.new(1,1,1)
			fpsLabel.Font=Enum.Font.Code
			fpsLabel.TextSize=13
			fpsLabel.BorderSizePixel=0
			Instance.new("UICorner",fpsLabel).CornerRadius=UDim.new(0,8)
			Instance.new("UIStroke",fpsLabel)
		end

		fpsLabel.Visible=true

		local count=0
		local last=os.clock()

		if fpsConn then fpsConn:Disconnect() end

		fpsConn=RS.RenderStepped:Connect(function()
			count+=1

			if os.clock()-last>=1 then
				local ping=0

				pcall(function()
					ping=math.floor(
						Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
					)
				end)

				fpsLabel.Text="FPS: "..count.."\nPING: "..ping.." ms"
				count=0
				last=os.clock()
			end
		end)
	else
		if fpsConn then
			fpsConn:Disconnect()
			fpsConn=nil
		end

		if fpsLabel then
			fpsLabel.Visible=false
		end
	end
end)

			if d.highlight then
				d.highlight.Enabled=espOn
				d.highlight.FillColor=c
				d.highlight.OutlineColor=Color3.new(1,1,1)
			end

			if d.line then
				if tracerOn then
					local pos,on=CG:WorldToViewportPoint(
						p.Character.HumanoidRootPart.Position
					)

					if on then
						d.line.From=Vector2.new(
							CG.ViewportSize.X/2,
							CG.ViewportSize.Y
						)
						d.line.To=Vector2.new(pos.X,pos.Y)
						d.line.Color=c
						d.line.Visible=true
					else
						d.line.Visible=false
					end
				else
					d.line.Visible=false
				end
			end
		end
	end
end)

local function drag(obj)
	local active=false
	local start
	local pos

	obj.InputBegan:Connect(function(i)
		if i.UserInputType==Enum.UserInputType.MouseButton1
		or i.UserInputType==Enum.UserInputType.Touch then
			active=true
			start=i.Position
			pos=obj.Position
		end
	end)

	UIS.InputChanged:Connect(function(i)
		if active and (
			i.UserInputType==Enum.UserInputType.MouseMovement
			or i.UserInputType==Enum.UserInputType.Touch
		) then
			local d=i.Position-start

			obj.Position=UDim2.new(
				pos.X.Scale,pos.X.Offset+d.X,
				pos.Y.Scale,pos.Y.Offset+d.Y
			)
		end
	end)

	UIS.InputEnded:Connect(function(i)
		if i.UserInputType==Enum.UserInputType.MouseButton1
		or i.UserInputType==Enum.UserInputType.Touch then
			active=false
		end
	end)
end
