local OriginalSettings = {}
local FpsBoostEnabled = false
local FpsConnection = nil
local FpsLabel = nil

local function ToggleBoostFPS(state)
    FpsBoostEnabled = state

    if FpsBoostEnabled then
        OriginalSettings.FogStart = Lighting.FogStart
        OriginalSettings.FogEnd = Lighting.FogEnd
        OriginalSettings.GlobalShadows = Lighting.GlobalShadows

        Lighting.FogStart = 0
        Lighting.FogEnd = 9e9
        Lighting.GlobalShadows = false

        for _, effect in pairs(Lighting:GetChildren()) do
            if effect:IsA("PostEffect") then
                effect.Enabled = false
            end
        end

        for _, obj in pairs(workspace:GetDescendants()) do
            if obj:IsA("BasePart") and not obj:IsA("Terrain") then
                obj.Material = Enum.Material.SmoothPlastic
            end
        end
    else
        if OriginalSettings.FogStart ~= nil then
            Lighting.FogStart = OriginalSettings.FogStart
            Lighting.FogEnd = OriginalSettings.FogEnd
            Lighting.GlobalShadows = OriginalSettings.GlobalShadows
        end
    end
end

local function ToggleFpsCounter(state)
    if state then
        if not FpsLabel then
            FpsLabel = Instance.new("TextLabel")
            FpsLabel.Name = "FPS_COUNTER"
            FpsLabel.Size = UDim2.new(0,140,0,45)
            FpsLabel.Position = UDim2.new(0.05,0,0.05,0)
            FpsLabel.BackgroundColor3 = Color3.fromRGB(15,15,15)
            FpsLabel.TextColor3 = Color3.fromRGB(255,255,255)
            FpsLabel.Font = Enum.Font.Code
            FpsLabel.TextSize = 14
            FpsLabel.BorderSizePixel = 0
            FpsLabel.ZIndex = 50
            FpsLabel.TextXAlignment = Enum.TextXAlignment.Center
            FpsLabel.Parent = sg

            local lbCorner = Instance.new("UICorner")
            lbCorner.CornerRadius = UDim.new(0,8)
            lbCorner.Parent = FpsLabel

            local fpsStroke = Instance.new("UIStroke")
            fpsStroke.Thickness = 1.8
            fpsStroke.Parent = FpsLabel

            local fpsCount = 0
            local lastUpdate = os.clock()

            FpsConnection = RS.RenderStepped:Connect(function()
                fpsCount += 1

                if os.clock() - lastUpdate >= 1 then
                    local ping = 0

                    pcall(function()
                        ping = math.floor(
                            Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
                        )
                    end)

                    FpsLabel.Text =
                        "FPS: " .. fpsCount ..
                        "\nPING: " .. ping .. " ms"

                    fpsCount = 0
                    lastUpdate = os.clock()
                end

                fpsStroke.Color =
                    Color3.fromHSV((os.clock() * 0.4) % 1,1,1)
            end)
        end

        FpsLabel.Visible = true
    else
        if FpsConnection then
            FpsConnection:Disconnect()
            FpsConnection = nil
        end

        if FpsLabel then
            FpsLabel.Visible = false
        end
    end
end

--==================================================
-- VISIBILITY CHECK
--==================================================
local function isVisible(targetPart)
    if not LP.Character
    or not LP.Character:FindFirstChild("HumanoidRootPart") then
        return false
    end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {
        LP.Character,
        targetPart.Parent
    }
    return workspace:Raycast(
        CG.CFrame.Position,
        targetPart.Position - CG.CFrame.Position,
        params
    ) == nil
end
--==================================================
-- CLEAN PLAYER
--==================================================
local function clean(p)
    if db[p] then
        if db[p].hl then
            db[p].hl:Destroy()
        end
        if db[p].li then
            db[p].li:Remove()
        end
        db[p] = nil
    end
end
--==================================================
-- ADD ESP
--==================================================
local function addESP(p)
    if p == LP then
        return
    end
    p.CharacterAdded:Connect(function(c)
        clean(p)
        if espOn then
            local line = Drawing.new("Line")
            line.Thickness = 1.5
            line.Transparency = 1
            line.Visible = false
            local highlight = Instance.new("Highlight")
            highlight.Parent = c
            highlight.FillTransparency = 0.45
            highlight.OutlineTransparency = 0
            db[p] = {
                hl = highlight,
                li = line
            }
        end
    end)
    p.CharacterRemoving:Connect(function()
        clean(p)
    end)
end
--==================================================
-- CLOSEST TARGET
--==================================================
local function getClosestInFOV()
    local target = nil
    local closest = fovCircle.Radius
    for _,v in pairs(P:GetPlayers()) do
        if v ~= LP
        and v.Character
        and v.Character:FindFirstChild(aimPart)
        and LP.Character
        and LP.Character:FindFirstChild("HumanoidRootPart") then
            local targetPart =
                v.Character[aimPart]
            local distance =
                (
                    LP.Character.HumanoidRootPart.Position
                    - targetPart.Position
                ).Magnitude
            if distance <= lockDistance then
                local pos,onScreen =
                    CG:WorldToViewportPoint(
                        targetPart.Position
                    )
                if onScreen then
                    local screenDistance =
                        (
                            Vector2.new(pos.X,pos.Y)
                            - fovCircle.Position
                        ).Magnitude
                    if screenDistance < closest
                    and isVisible(targetPart) then
                        closest = screenDistance
                        target = targetPart
                    end
                end
            end
        end
    end

end

-- MAIN RENDER LOOP
--==================================================

RS.RenderStepped:Connect(function()

    if not sg or not sg.Parent then
        return
    end

    local rgb =
        Color3.fromHSV(
            (os.clock() % 3) / 3,
            1,
            1
        )

    -- FOV POSITION
    fovCircle.Position =
        Vector2.new(
            CG.ViewportSize.X / 2,
            CG.ViewportSize.Y / 2
        )

    fovCircle.Color = rgb

    --==================================================
    -- AIM LOCK
    --==================================================

    if aimOn then

        local target =
            getClosestInFOV()

        if target then

            CG.CFrame =
                CFrame.new(
                    CG.CFrame.Position,
                    target.Position
                )

            if UIS.MouseBehavior ~= Enum.MouseBehavior.Default then

                UIS.MouseBehavior =
                    Enum.MouseBehavior.Default

            end

        end

    end

    --==================================================
    -- ESP / TRACERS UPDATE
    --==================================================

    for player,data in pairs(db) do

        if player
        and player.Character
        and player.Character:FindFirstChild("HumanoidRootPart") then

            -- ESP
            if espOn and data.hl then

                data.hl.Enabled = true

                data.hl.FillColor = rgb

                data.hl.OutlineColor =
                    Color3.fromRGB(
                        255,
                        255,
                        255
                    )

                data.hl.FillTransparency = 0.45

                data.hl.OutlineTransparency = 0

            elseif data.hl then

                data.hl.Enabled = false

            end

            -- TRACERS
            if trcOn and data.li then

                local root =
                    player.Character:FindFirstChild(
                        "HumanoidRootPart"
                    )

                if root then

                    local pos,onScreen =
                        CG:WorldToViewportPoint(
                            root.Position
                        )

                    if onScreen then

                        data.li.From =
                            Vector2.new(
                                CG.ViewportSize.X / 2,
                                CG.ViewportSize.Y
                            )

                        data.li.To =
                            Vector2.new(
                                pos.X,
                                pos.Y
                            )

                        data.li.Color = rgb

                        data.li.Thickness = 1.5

                        data.li.Visible = true

                    else

                        data.li.Visible = false

                    end

                else

                    data.li.Visible = false

                end

            elseif data.li then

                data.li.Visible = false

            end

        else

            if data.li then
                data.li.Visible = false
            end

            if data.hl then
                data.hl.Enabled = false
            end

        end

    end

end)

local function makeDraggable(guiObject)

    local dragging = false
    local dragInput = nil
    local dragStart = nil
    local startPos = nil

    guiObject.InputBegan:Connect(function(input)

        if input.UserInputType ==
            Enum.UserInputType.MouseButton1
        or input.UserInputType ==
            Enum.UserInputType.Touch then

            dragging = true

            dragStart =
                input.Position

            startPos =
                guiObject.Position

            input.Changed:Connect(function()

                if input.UserInputState ==
                    Enum.UserInputState.End then

                    dragging = false

                end

            end)

        end

    end)

    guiObject.InputChanged:Connect(function(input)

        if input.UserInputType ==
            Enum.UserInputType.MouseMovement
        or input.UserInputType ==
            Enum.UserInputType.Touch then

            dragInput = input

        end

    end)

    UIS.InputChanged:Connect(function(input)

        if input == dragInput
        and dragging
        and startPos
        and dragStart then

            local delta =
                input.Position - dragStart

            guiObject.Position =
                UDim2.new(
                    startPos.X.Scale,
                    startPos.X.Offset + delta.X,

                    startPos.Y.Scale,
                    startPos.Y.Offset + delta.Y
                )

        end

    end)

end

local function addPressAnimation(button)

    if not button then
        return
    end

    button.InputBegan:Connect(function(input)

        if input.UserInputType ==
            Enum.UserInputType.MouseButton1
        or input.UserInputType ==
            Enum.UserInputType.Touch then

            TS:Create(
                button,
                TweenInfo.new(
                    0.08,
                    Enum.EasingStyle.Quad,
                    Enum.EasingDirection.Out
                ),
                {
                    Size = UDim2.new(
                        button.Size.X.Scale,
                        button.Size.X.Offset - 2,
                        button.Size.Y.Scale,
                        button.Size.Y.Offset - 2
                    )
                }
            ):Play()

        end

    end)

    button.InputEnded:Connect(function(input)

        if input.UserInputType ==
            Enum.UserInputType.MouseButton1
        or input.UserInputType ==
            Enum.UserInputType.Touch then

            TS:Create(
                button,
                TweenInfo.new(
                    0.1,
                    Enum.EasingStyle.Back,
                    Enum.EasingDirection.Out
                ),
                {
                    Size = UDim2.new(
                        button.Size.X.Scale,
                        button.Size.X.Offset + 0,
                        button.Size.Y.Scale,
                        button.Size.Y.Offset + 0
                    )
                }
            ):Play()

        end

    end)

end
